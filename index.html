  </style>
</head>
<body>
  <h1>Ratio Split Calculator</h1>
  <label>

    <input type="number" id="total" placeholder="e.g. 100">
  </label>

  <label>
    Enter ratio (e.g. 3:2:5):<br>
    <input type="text" id="ratio" placeholder="e.g. 3:2:5">
  </label>

  <button onclick="calculateSplit()">Calculate</button>

  <div class="result" id="result"></div>

  <script>
    function calculateSplit() {
      const total = parseFloat(document.getElementById('total').value);
      const ratioInput = document.getElementById('ratio').value;
      const resultDiv = document.getElementById('result');

      if (isNaN(total) || total <= 0) {
        resultDiv.textContent = 'Please enter a valid positive number.';
        return;
      }

      const ratioParts = ratioInput.split(':').map(Number);

      if (ratioParts.some(isNaN) || ratioParts.length < 2) {
        resultDiv.textContent = 'Please enter a valid ratio like 3:2:5';
        return;
      }

      const sum = ratioParts.reduce((a, b) => a + b, 0);
      const splits = ratioParts.map(part => Math.round((part / sum) * total));

      resultDiv.innerHTML = splits.join(' : ');
    }
  </script>
</body>
